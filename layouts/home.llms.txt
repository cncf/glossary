{{- /*
  llms.txt index (https://llmstxt.org/), one per language at the site root.
  Lists every completed glossary term with a short definition, linking to the
  Markdown version of each page. Mirrors the sitemap's rule of publishing only
  pages whose status is Completed. URLs are absolute because llms.txt is copied
  and cached by tools that drop its source URL.

  Reference design: Docsy v0.17.0 theme/layouts/index.llms.txt (Apache-2.0),
  https://github.com/google/docsy/blob/v0.17.0/theme/layouts/index.llms.txt
  Docsy's index lists the home page, main menu, and doc sections. A glossary
  is only useful to an agent if the index lists the terms themselves, so this
  template is site-specific and stays in place after the Docsy upgrade.
*/ -}}
{{- $terms := slice -}}
{{- range .Site.RegularPages -}}
  {{- if eq (lower (strings.TrimSpace (printf "%v" (default "" .Params.status)))) "completed" -}}
    {{- $terms = $terms | append . -}}
  {{- end -}}
{{- end -}}
{{- $terms = sort $terms "RelPermalink" -}}
{{- $sections := slice -}}
{{- range .Site.Sections -}}
  {{- if eq (lower (strings.TrimSpace (printf "%v" (default "" .Params.status)))) "completed" -}}
    {{- $sections = $sections | append . -}}
  {{- end -}}
{{- end -}}
{{- $sections = sort $sections "RelPermalink" -}}
{{- $homeURL := .Permalink -}}
{{- with .OutputFormats.Get "markdown" }}{{ $homeURL = .Permalink }}{{ end -}}

- [{{ .Title | strings.TrimSpace }}]({{ $url }}){{ with $summary }}: {{ . }}{{ end }}
{{- end }}
